const POSView = () => {
  const [activeCategory, setActiveCategory] = useState('latte');
  const [cart, setCart] = useState([]);
  const [customerName, setCustomerName] = useState('');
  const [isMember, setIsMember] = useState(false);
  const [orderType, setOrderType] = useState('Dine In');

  const filteredProducts = mockProducts.filter(p => p.category === activeCategory);

  const addToCart = (product) => {
    setCart(prev => {
      const existing = prev.find(item => item.id === product.id);
      if (existing) {
        return prev.map(item => item.id === product.id ? { ...item, qty: item.qty + 1 } : item);
      }
      return [...prev, { ...product, qty: 1 }];
    });
  };

  const updateQty = (id, delta) => {
    setCart(prev => prev.map(item => {
      if (item.id === id) {
        const newQty = item.qty + delta;
        return newQty > 0 ? { ...item, qty: newQty } : item;
      }
      return item;
    }).filter(item => item.qty > 0));
  };

  const censorName = (name) => {
    if (!name) return 'Pelanggan Umum';
    if (!isMember) return name;
    
    const parts = name.split(' ');
    if (parts.length === 1) {
      return name.substring(0, 3) + '*'.repeat(Math.max(0, name.length - 3));
    }
    const first = parts[0];
    const rest = parts.slice(1).join(' ');
    return `${first} ${rest.charAt(0)}${'*'.repeat(Math.max(0, rest.length - 1))}`;
  };

  const checkMembership = () => {
    if (customerName.trim().length > 2) setIsMember(true);
  };

  const subtotal = cart.reduce((sum, item) => sum + (item.price * item.qty), 0);
  const tax = subtotal * 0.11;
  const total = subtotal + tax;
  const today = new Date().toLocaleDateString('id-ID', { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' });

  return (
    <div className="flex-1 flex overflow-hidden">
      {/* Area Utama POS (Kiri) */}
      <main className="flex-1 flex flex-col p-6 pr-0 overflow-hidden relative">
        <header className="flex justify-between items-center mb-8 pr-6 relative z-10">
          <div>
            <h1 className="font-gabriela text-2xl text-[#176637] mb-1">Kasir Sagara</h1>
            <p className="text-sm font-medium text-[#72AD43]">{today}</p>
          </div>
          <div className="flex items-center gap-6">
            <div className="relative">
              <Search className="absolute left-4 top-1/2 -translate-y-1/2 text-[#176637]/40" size={18} />
              <input type="text" placeholder="Cari menu (⌘K)" className="bg-white border-2 border-[#176637]/10 rounded-full pl-12 pr-4 py-2 w-64 text-sm focus:outline-none focus:border-[#72AD43] transition-colors text-[#176637]" />
            </div>
          </div>
        </header>

        {/* Kategori Menu */}
        <div className="flex gap-4 overflow-x-auto hide-scroll pb-2 mb-6 pr-6 relative z-10">
          {categories.map((cat) => {
            const isActive = activeCategory === cat.id;
            return (
              <button
                key={cat.id}
                onClick={() => setActiveCategory(cat.id)}
                className={`flex-shrink-0 relative overflow-hidden flex flex-col justify-between p-5 w-44 h-24 text-left transition-all duration-300 ${
                  isActive 
                    ? 'bg-[#176637] text-[#FFF6DB] rounded-tr-[30px] rounded-bl-[30px] rounded-tl-xl rounded-br-xl shadow-[4px_4px_0px_#72AD43]' 
                    : 'bg-white text-[#176637] border-2 border-[#176637]/10 rounded-tr-[30px] rounded-bl-[30px] rounded-tl-xl rounded-br-xl hover:border-[#72AD43]'
                }`}
              >
                <div className="relative z-10 flex justify-between items-center w-full">
                  <h3 className="font-gabriela text-lg leading-tight">{cat.name}</h3>
                  {cat.alert && <span className="bg-red-500 w-2 h-2 rounded-full"></span>}
                </div>
                <p className="text-xs opacity-80 relative z-10">{cat.count} items</p>
                {isActive && (
                  <svg className="absolute -right-2 -bottom-2 opacity-20 w-20 h-20" viewBox="0 0 100 100" fill="#FFF6DB">
                    <path d="M10,90 C10,50 30,20 60,10 C80,30 50,60 40,80 C30,100 20,95 10,90 Z" />
                  </svg>
                )}
              </button>
            );
          })}
        </div>

        {/* Grid Produk */}
        <div className="flex-1 overflow-y-auto pr-6 pb-6 relative z-10">
          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
            {filteredProducts.map((product) => (
              <div key={product.id} className="bg-white p-4 flex flex-col h-56 border border-[#176637]/10 hover:border-[#72AD43] hover:shadow-[4px_4px_0px_#176637] transition-all duration-300 group rounded-tr-[30px] rounded-bl-[30px] rounded-tl-lg rounded-br-lg">
                <div className="w-full h-24 bg-[#FFF6DB]/50 rounded-tl-xl rounded-br-xl mb-3 flex items-center justify-center text-4xl group-hover:scale-105 transition-transform">
                  {product.img}
                </div>
                <h4 className="font-semibold text-sm text-[#176637] leading-tight mb-1 flex-1">{product.name}</h4>
                <div className="flex justify-between items-end mt-auto">
                  <span className="font-bold text-[#FF901A] text-sm">Rp {product.price.toLocaleString('id-ID')}</span>
                  <button onClick={() => addToCart(product)} className="w-7 h-7 rounded-full border-2 border-[#176637] text-[#176637] flex items-center justify-center hover:bg-[#176637] hover:text-[#FFF6DB] transition-colors">
                    <Plus size={14} />
                  </button>
                </div>
              </div>
            ))}
          </div>
        </div>
      </main>

      {/* Cart Aside (Kanan) */}
      <aside className="w-[380px] bg-white shadow-[-10px_0_30px_rgba(23,102,55,0.05)] rounded-tl-[40px] flex flex-col h-full border-l border-[#176637]/10 relative z-20">
        <div className="p-6 border-b border-[#176637]/5 flex justify-between items-center bg-[#FFF6DB]/30 rounded-tl-[40px]">
          <div className="text-center w-full">
            <h2 className="font-gabriela text-lg font-bold text-[#176637]">Nota Pesanan</h2>
            <p className="text-xs text-[#176637]/60">#TRX-88291</p>
          </div>
        </div>

        <div className="p-6 flex-1 flex flex-col overflow-hidden">
          <div className="flex bg-[#FFF6DB] rounded-full p-1 mb-5 border border-[#176637]/10">
            {['Dine In', 'Take Away', 'Delivery'].map(type => (
              <button 
                key={type} onClick={() => setOrderType(type)}
                className={`flex-1 text-xs py-2 rounded-full font-medium transition-colors ${
                  orderType === type ? 'bg-[#176637] text-[#FFF6DB] shadow-sm' : 'text-[#176637]/70 hover:text-[#176637]'
                }`}
              >
                {type}
              </button>
            ))}
          </div>

          <div className="flex gap-2 mb-5">
            <input 
              type="text" placeholder="Nama Pelanggan" value={customerName} onChange={(e) => setCustomerName(e.target.value)}
              className="flex-1 border-2 border-[#176637]/10 rounded-xl px-3 py-2 text-sm focus:outline-none focus:border-[#72AD43] text-[#176637]"
            />
            <button onClick={checkMembership} className={`px-3 rounded-xl text-xs font-bold border-2 transition-all ${isMember ? 'bg-[#72AD43]/20 border-[#72AD43] text-[#176637]' : 'border-[#FF901A] text-[#FF901A] hover:bg-[#FF901A] hover:text-[#FFF6DB]'}`}>
              {isMember ? '✓ Member' : 'Cek'}
            </button>
          </div>

          <div className="flex-1 overflow-y-auto hide-scroll mb-6">
            {cart.length === 0 ? (
              <div className="h-full flex flex-col items-center justify-center text-[#176637]/30">
                <Coffee size={32} className="mb-2" />
                <p className="text-sm">Belum ada pesanan</p>
              </div>
            ) : (
              <div className="space-y-3">
                {cart.map(item => (
                  <div key={item.id} className="flex items-center gap-3 bg-[#FFF6DB]/20 p-2 rounded-xl border border-[#FFF6DB]">
                    <div className="w-10 h-10 bg-white rounded-lg flex items-center justify-center text-xl shadow-sm">{item.img}</div>
                    <div className="flex-1 overflow-hidden">
                      <h4 className="text-sm font-semibold text-[#176637] truncate">{item.name}</h4>
                      <p className="text-xs text-[#176637]/60">Rp {item.price.toLocaleString('id-ID')}</p>
                    </div>
                    <div className="flex items-center gap-2 bg-white px-1.5 py-1 rounded-lg border border-[#176637]/10">
                      <button onClick={() => updateQty(item.id, -1)} className="text-[#176637]"><Minus size={12} /></button>
                      <span className="text-xs font-bold w-3 text-center">{item.qty}</span>
                      <button onClick={() => updateQty(item.id, 1)} className="text-[#176637]"><Plus size={12} /></button>
                    </div>
                  </div>
                ))}
              </div>
            )}
          </div>

          <div className="mt-auto border-t-2 border-dashed border-[#176637]/10 pt-4">
            {customerName && (
              <div className="bg-[#176637]/5 p-2.5 rounded-lg mb-3 border border-[#176637]/10 flex justify-between items-center">
                <span className="text-xs text-[#176637]/70">Pelanggan:</span>
                <span className="text-sm font-bold text-[#176637] font-gabriela">{censorName(customerName)}</span>
              </div>
            )}
            <div className="space-y-1.5 mb-4">
              <div className="flex justify-between text-sm text-[#176637]/70">
                <span>Subtotal</span><span>Rp {subtotal.toLocaleString('id-ID')}</span>
              </div>
              <div className="flex justify-between text-sm text-[#176637]/70">
                <span>Pajak (11%)</span><span>Rp {tax.toLocaleString('id-ID')}</span>
              </div>
              <div className="flex justify-between text-lg font-bold text-[#176637] pt-2 border-t border-[#176637]/10 mt-2">
                <span>Total</span><span>Rp {total.toLocaleString('id-ID')}</span>
              </div>
            </div>
            <button 
              className={`w-full py-3 rounded-xl font-bold flex items-center justify-between px-5 transition-all shadow-[4px_4px_0px_#72AD43] ${cart.length > 0 ? 'bg-[#176637] text-[#FFF6DB] hover:-translate-y-1' : 'bg-gray-300 text-gray-500 cursor-not-allowed shadow-none'}`}
              disabled={cart.length === 0}
            >
              <span>Bayar</span>
              <span className="bg-[#FFF6DB]/20 px-2 py-1 rounded-lg text-sm">Rp {total.toLocaleString('id-ID')}</span>
            </button>
          </div>
        </div>
      </aside>
    </div>
  );
};

export default function App() {
  const [currentRole, setCurrentRole] = useState('investor');
  const [activeTab, setActiveTab] = useState('investor_dashboard');

  const handleRoleSwitch = () => {
    if (currentRole === 'mitra') {
      setCurrentRole('investor');
      setActiveTab('investor_dashboard');
    } else {
      setCurrentRole('mitra');
      setActiveTab('pos');
    }
  };

  const getNavItems = () => {
    if (currentRole === 'investor') {
      return [
        { id: 'investor_dashboard', icon: LayoutDashboard, label: 'Dashboard Investor' },
      ];
    }
    return [
      { id: 'dashboard', icon: LayoutDashboard, label: 'Dashboard Outlet' },
      { id: 'pos', icon: Store, label: 'Point of Sale' },
      { id: 'employees', icon: Users, label: 'Karyawan' },
      { id: 'supply', icon: Package, label: 'Inventaris' },
    ];
  };

  const navItems = getNavItems();

  return (
    <>
      <GlobalStyles />
      <div className="flex h-screen bg-[#FFF6DB] overflow-hidden">
        
        {/* Sidebar Navigasi Dinamis */}
        <aside className="w-20 bg-[#176637] h-screen flex flex-col items-center py-6 shadow-xl relative z-30">
          <div className="w-12 h-12 border-2 border-[#FFF6DB]/30 border-dashed rounded flex items-center justify-center text-[10px] text-[#FFF6DB] text-center leading-tight mb-4 font-gabriela">
            Sagara<br/>Logo
          </div>

          {/* Tombol Demo Switch Role (Investor <-> Mitra) */}
          <button 
            onClick={handleRoleSwitch}
            title="Ganti Role (Demo)"
            className="mb-8 px-2 py-1 bg-[#FF901A] text-[#FFF6DB] text-[9px] font-bold rounded uppercase tracking-wider hover:scale-105 transition-transform text-center leading-tight"
          >
            Role:<br/>{currentRole === 'mitra' ? 'Mitra' : 'Investor'}
          </button>
          
          <nav className="flex flex-col gap-4 w-full">
            {navItems.map(item => {
              const isActive = activeTab === item.id;
              return (
                <button 
                  key={item.id}
                  onClick={() => setActiveTab(item.id)}
                  title={item.label}
                  className={`w-full flex justify-center py-4 relative transition-all ${
                    isActive 
                      ? 'text-[#FFF6DB] bg-[#FFF6DB]/10 border-r-4 border-[#FF901A]' 
                      : 'text-[#FFF6DB]/50 hover:text-[#FFF6DB] hover:bg-[#FFF6DB]/5'
                  }`}
                >
                  <item.icon size={24} className={isActive ? 'animate-bounce-short' : ''} />
                </button>
              )
            })}
          </nav>
          
          <div className="mt-auto flex flex-col gap-4 w-full">
            <button className="w-full flex justify-center text-[#FFF6DB]/50 hover:text-[#FFF6DB] transition-colors py-2"><Settings size={22} /></button>
            <button className="w-full flex justify-center text-[#FFF6DB]/50 hover:text-[#FF901A] transition-colors py-2"><LogOut size={22} /></button>
          </div>
        </aside>

        {/* Dynamic Content Area based on Sidebar Selection */}
        {activeTab === 'dashboard' && <DashboardView />}
        {activeTab === 'pos' && <POSView />}
        {activeTab === 'employees' && <EmployeeView />}
        {activeTab === 'supply' && <SupplyView />}
        {activeTab === 'investor_dashboard' && <InvestorView />}
        
      </div>
    </>
  );
}
